// ذخیره محصول (افزودن یا ویرایش) function saveProduct() { const id = document.getElementById('productId').value; const product = { id: id ? parseInt(id) : Date.now(), code: document.getElementById('productCode').value, name: document.getElementById('productName').value, category: document.getElementById('productCategory').value, unit: document.getElementById('productUnit').value, purchasePrice: parseInt(document.getElementById('purchasePrice').value) || 0, salePrice: parseInt(document.getElementById('salePrice').value) || 0, stock: parseInt(document.getElementById('initialStock').value) || 0, minStock: parseInt(document.getElementById('minStock').value) || 0, description: document.getElementById('productDescription').value, status: document.getElementById('productStatus').value, barcode: document.getElementById('productBarcode').value, image: '' }; // اعتبارسنجی if (!product.code || !product.name || !product.category || !product.purchasePrice || !product.salePrice) { showToast('لطفاً تمام فیلدهای الزامی را پر کنید', 'error'); return; } if (product.salePrice < product.purchasePrice) { showToast('قیمت فروش نمی‌تواند از قیمت خرید کمتر باشد', 'error'); return; } // بررسی تکراری نبودن کد محصول if (!id && productsData.some(p => p.code === product.code)) { showToast('کد محصول تکراری است', 'error'); return; } // پردازش تصویر const imageFile = document.getElementById('productImage').files[0]; if (imageFile) { const reader = new FileReader(); reader.onload = (e) => { product.image = e.target.result; saveProductToArray(product, id); }; reader.readAsDataURL(imageFile); } else if (id) { const existingProduct = productsData.find(p => p.id === parseInt(id)); product.image = existingProduct?.image || ''; saveProductToArray(product, id); } else { saveProductToArray(product, id); } } function saveProductToArray(product, id) { if (id) { const index = productsData.findIndex(p => p.id === parseInt(id)); if (index !== -1) { productsData[index] = { ...productsData[index], ...product }; } showToast('کالا با موفقیت ویرایش شد', 'success'); } else { productsData.push(product); showToast('کالا با موفقیت اضافه شد', 'success'); } renderProductsTable(); bootstrap.Modal.getInstance(document.getElementById('productModal')).hide(); resetProductForm(); } function resetProductForm() { document.getElementById('productId').value = ''; document.getElementById('modalTitle').innerText = 'افزودن کالا'; document.getElementById('productForm').reset(); document.getElementById('imagePreview').innerHTML = ''; document.getElementById('initialStock').value = '0'; document.getElementById('minStock').value = '5'; currentProductId = null; } function showDeleteModal(id, name) { deleteProductId = id; document.getElementById('deleteProductName').innerText = name; new bootstrap.Modal(document.getElementById('deleteModal')).show(); } function confirmDelete() { if (deleteProductId) { productsData = productsData.filter(p => p.id !== deleteProductId); renderProductsTable(); showToast('کالا با موفقیت حذف شد', 'success'); bootstrap.Modal.getInstance(document.getElementById('deleteModal')).hide(); deleteProductId = null; } } // پیش‌نمایش تصویر document.getElementById('productImage')?.addEventListener('change', (e) => { const file = e.target.files[0]; const preview = document.getElementById('imagePreview'); if (file && preview) { const reader = new FileReader(); reader.onload = (ev) => { preview.innerHTML = ``; }; reader.readAsDataURL(file); } }); // نمونه fetch برای اتصال به API واقعی async function fetchProductsFromAPI() { try { const response = await fetch('/api/products'); if (response.ok) { const data = await response.json(); productsData = data; renderProductsTable(); } } catch (error) { console.log('Using mock data'); loadProducts(); } } async function saveProductToAPI(product) { try { const formData = new FormData(); Object.keys(product).forEach(key => formData.append(key, product[key])); const response = await fetch('/api/products', { method: product.id ? 'PUT' : 'POST', body: formData }); if (response.ok) { showToast('کالا با موفقیت ذخیره شد', 'success'); fetchProductsFromAPI(); } } catch (error) { console.error('API Error:', error); } } // رویدادها document.getElementById('saveProductBtn')?.addEventListener('click', saveProduct); document.getElementById('confirmDeleteBtn')?.addEventListener('click', confirmDelete); document.getElementById('addProductBtn')?.addEventListener('click', () => { resetProductForm(); }); // مقداردهی اولیه loadSidebarAndNavbar('products'); loadProducts(); // برای استفاده از API واقعی، خط زیر را جایگزین loadProducts() کنید // fetchProductsFromAPI();
